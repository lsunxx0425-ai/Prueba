\begin{enumerate}[2)]
\item $x^3-x^2+2x-2<0$ \\ \\
$x^3-x^2+2x-2<0 \Longleftrightarrow (x-1)(x^2+2)<0$
\begin{enumerate}[a)]
\item $x-1<0$ i $x^2+2>0\Longleftrightarrow x<1$ i $x>\pm \sqrt{-2}$ \\
Com que $\sqrt{-2}$ no existeix, per tant $x\in (-\infty, 1)$
\item $x-1<0$ i $x^2+2<0$ \\
No pot ser, perquè $x^2+2$ sempre és positiva.
\end{enumerate}
Llavors, $x\in (-\infty, 1)$\\
\end{enumerate}
