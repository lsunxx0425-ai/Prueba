\begin{enumerate}[1)]
\item $\vert x^2-1 \vert <x+1$ \\ \\
$\vert x^2-1 \vert <x+1 \Longleftrightarrow -(x+1)<x^2-1<x+1 \Longleftrightarrow -x<x^2<x+2$ 
\begin{enumerate}[a)]
\item $-x<x^2 \Longleftrightarrow 0<x^2+x$ \\
Suposem que $0=x^2+x \Longleftrightarrow 0=(x+1)x$, llavors tenim $x=0$ i $x+1=0 \Longleftrightarrow x=-1$ \\
$x\in (-\infty, -1)\cup(0,+\infty)$
\item $x^2<x+2 \Longleftrightarrow x^2-x-2<0$ \\
Suposem que $x^2-x-2=0$, aplicam la fòrmula de segon grau i tenim \begin{equation*}
x = \frac{1 \pm \sqrt{(-1)^2- 4\cdot 1(-2)}}
{2\cdot 1} = \frac{1 \pm 3}{2} 
\end{equation*}
Per a $x=\frac{1+3}{2}$ tenim $x=2$, i per a $x=\frac{1-3}{2}$ tenim $x=-1$. \\
$x\in (-1, 2)$ 
\end{enumerate}
Llavors, $x\in ((-\infty, -1)\cup(0,+\infty))\cap (-1, 2)= (0, 2)$\\
\end{enumerate}
